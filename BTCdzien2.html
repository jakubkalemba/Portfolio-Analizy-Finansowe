# 1. Wyciągamy samego Bitcoina do nowej zmiennej, żeby było czytelniej
btc = dane['BTC-USD'].to_frame() # .to_frame() zamienia to w ładną tabelkę

btc['SMA200'] = btc['BTC-USD'].rolling(window=200).mean()
