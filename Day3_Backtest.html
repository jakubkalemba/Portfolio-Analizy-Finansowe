import yfinance as yf
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ==========================================
# 1. POBIERANIE DANYCH (WERSJA NAPRAWIONA)
# ==========================================
print(">>> Pobieram dane dla Bitcoina...")
# Pobieramy dane
df = yf.download(['BTC-USD'], start="2019-01-01", end="2025-01-01")

# MAGICZNA NAPRAWA STRUKTURY:
# Niezależnie co zwróci yfinance, wyciągamy kolumnę 'Close' (Zamknięcie)
try:
    # Próba 1: Jeśli 'Close' jest główną kolumną (MultiIndex)
    df = df['Close']
except:
    # Próba 2: Jeśli coś poszło inaczej, bierzemy po prostu kolumnę o nazwie 'Close'
    # (to zabezpieczenie na różne wersje biblioteki)
    df = df.xs('Close', axis=1, level=0)

# Upewniamy się, że to tabelka (DataFrame), a nie lista (Series)
if isinstance(df, pd.Series):
    df = df.to_frame()

# Zmieniamy nazwę kolumny na 'Cena' (żeby reszta kodu działała)
df.columns = ['Cena']

print(">>> Dane gotowe! Pierwsze 5 wierszy:")
print(df.head())

# ==========================================
# 2. OBLICZENIA I STRATEGIA (Reszta bez zmian)
# ==========================================
df['SMA_50'] = df['Cena'].rolling(window=50).mean()
df['SMA_200'] = df['Cena'].rolling(window=200).mean()

# Logika handlu: 1 = KUP, 0 = SPRZEDAJ (Gotówka)
df['Sygnał'] = np.where(df['SMA_50'] > df['SMA_200'], 1, 0)
df['Pozycja'] = df['Sygnał'].shift(1) # Przesuwamy o 1 dzień (realizm)

# Liczenie zysków
df['Zwrot_Rynku'] = df['Cena'].pct_change()
df['Zwrot_Strategii'] = df['Zwrot_Rynku'] * df['Pozycja']

# Symulacja 1000 PLN
kapital_startowy = 1000
df['Portfel_HODL'] = (1 + df['Zwrot_Rynku']).cumprod() * kapital_startowy
df['Portfel_Strategia'] = (1 + df['Zwrot_Strategii']).cumprod() * kapital_startowy

# ==========================================
# 3. WYNIKI
# ==========================================
print("-" * 30)
print(f"Kapitał końcowy (Kup i Trzymaj): {df['Portfel_HODL'].iloc[-1]:.2f} PLN")
print(f"Kapitał końcowy (Twoja Strategia): {df['Portfel_Strategia'].iloc[-1]:.2f} PLN")
print("-" * 30)

plt.figure(figsize=(12, 6))
plt.plot(df['Portfel_HODL'], label='Kup i Trzymaj (HODL)', color='gray', alpha=0.5)
plt.plot(df['Portfel_Strategia'], label='Strategia SMA 50/200', color='green', linewidth=2)
plt.title('Bitcoin: Strategia Algorytmiczna vs HODL')
plt.ylabel('Wartość portfela (PLN)')
plt.legend()
plt.grid(True)
plt.show()
