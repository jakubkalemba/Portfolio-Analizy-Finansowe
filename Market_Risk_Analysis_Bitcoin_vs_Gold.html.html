# 1. Obliczamy dzienne stopy zwrotu (o ile % cena zmieniła się z dnia na dzień)
# To jest kluczowa funkcja w finansach: pct_change()
zwroty_dzienne = dane.pct_change()

# 2. Symulacja: Jak rosłaby inwestycja 1000 PLN?
# cumprod() to skumulowany iloczyn - idealny do liczenia procentu składanego
kapital_startowy = 1000
wartosc_portfela = (1 + zwroty_dzienne).cumprod() * kapital_startowy

# 3. Rysujemy wynik
plt.figure(figsize=(12, 6))
plt.plot(wartosc_portfela)
plt.title('Symulacja inwestycji 1000 PLN: Bitcoin vs Złoto')
plt.ylabel('Wartość w PLN')
plt.grid(True)
plt.show()

# 4. Wyświetl wynik na dziś (ostatni dzień)
print("Wartość końcowa inwestycji (z 1000 PLN):")
print(wartosc_portfela.iloc[-1].round(2))
